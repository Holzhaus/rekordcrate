84/// Insert the default color rows into the database.
85pub fn insert_default_colors<RW: Read + Write + Seek>(db: &mut Database<RW>) -> crate::Result<()> {
86    for &(id, ref color, name) in DEFAULT_COLORS {
87        db.add_row(Color {
88            unknown1: 0,
89            unknown2: id,
90            color: color.clone(),
91            unknown3: 0,
92            name: DeviceSQLString::new(name)?,
93        })?;
94    }
95    Ok(())
96}
97
98/// Insert the default column rows into the database.
99pub fn insert_default_columns<RW: Read + Write + Seek>(db: &mut Database<RW>) -> crate::Result<()> {
100    for &(id, unknown0, name) in DEFAULT_COLUMNS {
101        db.add_row(ColumnEntry {
102            id,
103            unknown0,
104            column_name: DeviceSQLString::new(name)?,
105        })?;
106    }
107    Ok(())
108}
109
110/// Insert the default menu rows into the database.
111pub fn insert_default_menus<RW: Read + Write + Seek>(db: &mut Database<RW>) -> crate::Result<()> {
112    for &(category_id, content_pointer, unknown, ref visibility, sort_order) in DEFAULT_MENUS {
113        db.add_row(Menu {
114            category_id,
115            content_pointer,
116            unknown,
117            visibility: *visibility,
118            sort_order,
119        })?;
120    }
121    Ok(())
122}
