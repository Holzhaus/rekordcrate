9//! The [`Track`] type: a track as a user thinks of it, used both to add tracks to a
10//! [`DeviceExport`](crate::DeviceExport) and to read them back.
11
12use crate::pdb::offset_array::OffsetArrayContainer;
13use crate::pdb::string::DeviceSQLString;
14use crate::pdb::{
15    AlbumId, ArtistId, ArtworkId, GenreId, KeyId, LabelId, PageHeapObject, Subtype,
16    Track as PdbTrack, TrackId, TrackStrings,
17};
18use crate::util::{ColorIndex, FileType, MaybeCalculated, Rating};
19use crate::Result;

21/// Minimum allocated (4-byte-aligned) size of a Track row, in bytes. A CDJ-350 crashes entering
22/// its "TRACK" menu when any track row is shorter than this; 221 was the smallest value found to
23/// avoid the crash (by trial and error). [`DeviceExport::add_track`](crate::DeviceExport::add_track)
24/// pads `comment` to satisfy it; the low-level [`pdb`](crate::pdb) layer does not enforce it.
25pub(crate) const MIN_TRACK_ROW_SIZE: u16 = 221;
26
27/// The string value of the PDB's boolean-like string fields when they are on.
28pub(crate) const ON: &str = "ON";
29
30/// A track as a user thinks of it: plain `String` fields, no foreign-key IDs. An empty string
31/// means "none", as in the PDB itself.
32///
33/// [`DeviceExport::add_track`](crate::DeviceExport::add_track) takes one and
34/// [`DeviceExport::tracks`](crate::DeviceExport::tracks) hands them back. Fields not exposed
35/// here need a `pdb::Track` row driven directly via [`crate::pdb::io::Database`].
36#[derive(Debug, Clone, Default, PartialEq)]
37pub struct Track {
38    /// Track title.
39    pub title: String,
40    /// Performing artist name.
41    pub artist: String,
42    /// Album name.
43    pub album: String,
44    /// Genre name.
45    pub genre: String,
46    /// Musical key name (e.g. "Am", "F#m", "8A"), stored as given.
47    pub key: String,
48    /// Record label name.
49    pub label: String,
50    /// Composer name.
51    pub composer: String,
52    /// Remixer name.
53    pub remixer: String,
54    /// Original performer, distinct from `artist` (covers/reworks).
55    pub orig_artist: String,
56    /// Free-text comment. Padded with trailing spaces when the row would fall under the minimum
57    /// size some players need; trailing spaces are trimmed when reading back.
58    pub comment: String,
59    /// ISRC, in rekordbox's mangled format.
60    pub isrc: String,
61    /// Lyricist name.
62    pub lyricist: String,
63    /// Remix/mix name.
64    pub mix_name: String,
65    /// Free text. Rekordbox itself writes strict `YYYY-MM-DD` or empty here, but this field is
66    /// passed through verbatim so callers can probe hardware behavior with arbitrary strings.
67    pub release_date: String,
68    /// Free text, commonly `YYYY-MM-DD` (see [`Self::release_date`]).
69    pub date_added: String,
70    /// Device-relative file path (e.g. `/Contents/Artist - Title.mp3`). Dedup key when non-empty.
71    pub file_path: String,
72    /// File name without path.
73    pub filename: String,
74    /// Device path of the artwork thumbnail (e.g. `/PIONEER/Artwork/00001/a1.jpg`), stored
75    /// verbatim in the `Artwork` row. Empty = none. The caller owns placing the image files it
76    /// names — or, with the `artwork` feature, gets a path whose files the export writes itself
77    /// from `DeviceExport::import_artwork`.
78    pub artwork_path: String,
79    /// Track "message" field shown in Rekordbox.
80    pub message: String,
81    /// Tempo in BPM; encoded to centi-BPM (× 100) in the PDB.
82    pub tempo: f32,
83    /// Bitrate in kbps.
84    pub bitrate: u32,
85    /// Sample rate in Hz.
86    pub sample_rate: u32,
87    /// Bits per sample of the audio file.
88    pub sample_depth: u16,
89    /// Playback duration in seconds. PDB field is `u16` (ceiling ~18.2 h).
90    pub duration_secs: u16,
91    /// File size in bytes.
92    pub file_size: u32,
93    /// Track number within the album.
94    pub track_number: u32,
95    /// Disc number.
96    pub disc_number: u16,
97    /// Release year.
98    pub year: u16,
99    /// Number of times the track was played.
100    pub play_count: u16,
101    /// Star rating.
102    pub rating: Rating,
103    /// Color label.
104    pub color: ColorIndex,
105    /// Audio file format.
106    pub file_type: FileType,
107    /// Whether stored hotcues auto-load on a CDJ. Maps to the PDB string `"ON"` / empty.
108    pub autoload_hotcues: bool,
109}
110
111/// Allocated size of a Track row on the page heap: its serialized size, 4-byte aligned.
112fn allocated_row_size(row: &PdbTrack) -> u16 {
113    row.heap_bytes_required(()).next_multiple_of(4)
114}
115
116impl Track {
117    /// The foreign-key ids (and the track's own id) are left at 0 for the caller to fill.
118    pub(crate) fn to_row(&self) -> Result<PdbTrack> {
119        let track = self;
120        // Only `comment` grows below; hoist the rest so they're encoded once.
121        let isrc = DeviceSQLString::new(&track.isrc)?;
122        let lyricist = DeviceSQLString::new(&track.lyricist)?;
123        let message = DeviceSQLString::new(&track.message)?;
124        let mix_name = DeviceSQLString::new(&track.mix_name)?;
125        let release_date = DeviceSQLString::new(&track.release_date)?;
126        let date_added = DeviceSQLString::new(&track.date_added)?;
127        let title = DeviceSQLString::new(&track.title)?;
128        let filename = DeviceSQLString::new(&track.filename)?;
129        let file_path = DeviceSQLString::new(&track.file_path)?;
130        let autoload_hotcues = if track.autoload_hotcues {
131            DeviceSQLString::new(ON)?
132        } else {
133            DeviceSQLString::empty()
134        };
135
136        let mut pdb_track = PdbTrack {
137            // Reverse-engineered constants observed on fresh Rekordbox track rows.
138            subtype: Subtype(0x24),
139            index_shift: 0,
140            bitmask: 788_224,
141            unknown5: 41,
142            sample_rate: track.sample_rate,
143            sample_depth: track.sample_depth,
144            bitrate: track.bitrate,
145            duration: track.duration_secs,
146            file_size: track.file_size,
147            tempo: (track.tempo * 100.0).round() as u32,
148            file_type: track.file_type.clone(),
149            track_number: track.track_number,
150            disc_number: track.disc_number,
151            year: track.year,
152            play_count: track.play_count,
153            rating: track.rating.to_byte(),
154            color: track.color.clone(),
155            composer_id: ArtistId(0),
156            artwork_id: ArtworkId(0),
157            key_id: KeyId(0),
158            orig_artist_id: ArtistId(0),
159            label_id: LabelId(0),
160            remixer_id: ArtistId(0),
161            genre_id: GenreId(0),
162            album_id: AlbumId(0),
163            artist_id: ArtistId(0),
164            id: TrackId(0),
165            unknown2: 0,
166            unknown3: 0,
167            unknown4: 0,
168            offsets: OffsetArrayContainer {
169                offsets: MaybeCalculated::Calculated,
170                inner: TrackStrings {
171                    isrc,
172                    lyricist,
173                    // TODO(acrilique): "1" is only the most common value on
174                    // current fixtures. The real meaning behind these fields
175                    // is still TBD.
176                    unknown_string2: DeviceSQLString::new("1")?,
177                    unknown_string3: DeviceSQLString::new("1")?,
178                    unknown_string4: DeviceSQLString::empty(),
179                    message,
180                    // TODO(acrilique): this is not constant on real exports.
181                    // Maybe some users of the library will want to be able to
182                    // choose the value for it...
183                    publish_track_information: DeviceSQLString::new(ON)?,
184                    autoload_hotcues,
185                    unknown_string5: DeviceSQLString::empty(),
186                    unknown_string6: DeviceSQLString::empty(),
187                    date_added,
188                    release_date,
189                    mix_name,
190                    unknown_string7: DeviceSQLString::empty(),
191                    analyze_path: DeviceSQLString::empty(),
192                    analyze_date: DeviceSQLString::empty(),
193                    comment: DeviceSQLString::empty(),
194                    title,
195                    unknown_string8: DeviceSQLString::empty(),
196                    filename,
197                    file_path,
198                },
199            },
200        };
201
202        // `comment` absorbs the padding up to `MIN_TRACK_ROW_SIZE`: trailing spaces are harmless
203        // in free text.
204        let mut comment = track.comment.clone();
205        loop {
206            pdb_track.offsets.inner.comment = DeviceSQLString::new(&comment)?;
207            if allocated_row_size(&pdb_track) >= MIN_TRACK_ROW_SIZE {
208                break;
209            }
210            comment.push(' ');
211        }
212
213        Ok(pdb_track)
214    }
