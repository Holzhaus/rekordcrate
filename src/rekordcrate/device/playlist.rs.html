9//! The playlist tree of a device export.
10
11use crate::pdb::io::Database;
12use crate::pdb::{PlaylistTreeNode, PlaylistTreeNodeId};
13use fallible_iterator::FallibleIterator;
14use std::collections::HashMap;
15use std::io::{Read, Seek};
16
17/// Either a playlist folder or a playlist.
18#[derive(Debug, PartialEq)]
19pub enum PlaylistNode {
20    /// A folder containing child [`PlaylistNode`]s.
21    Folder(PlaylistFolder),
22    /// A playlist (leaf).
23    Playlist(Playlist),
24}
25
26/// A playlist folder.
27#[derive(Debug, PartialEq)]
28pub struct PlaylistFolder {
29    /// ID of this node in the playlist tree.
30    pub id: PlaylistTreeNodeId,
31    /// Name of the playlist folder.
32    pub name: String,
33    /// Child nodes of the playlist folder.
34    pub children: Vec<PlaylistNode>,
35}
36
37/// A playlist.
38#[derive(Debug, PartialEq, Eq)]
39pub struct Playlist {
40    /// ID of this node in the playlist tree.
41    pub id: PlaylistTreeNodeId,
42    /// Name of the playlist.
43    pub name: String,
44}
45
46/// Build the playlist tree from the PDB database, each level in display order.
47pub fn get_playlists<R: Read + Seek>(db: &mut Database<R>) -> crate::Result<Vec<PlaylistNode>> {
48    let mut playlists: HashMap<PlaylistTreeNodeId, Vec<PlaylistTreeNode>> = HashMap::new();
49
50    db.iter_rows::<PlaylistTreeNode>()?.for_each(|node| {
51        playlists
52            .entry(node.parent_id)
53            .or_default()
54            .push(node.clone());
55        Ok(())
56    })?;
57    // Rows are stored in no particular order; `sort_order` is the display order among siblings.
58    for siblings in playlists.values_mut() {
59        siblings.sort_by_key(|node| node.sort_order);
60    }
61
62    fn get_child_nodes(
63        playlists: &HashMap<PlaylistTreeNodeId, Vec<PlaylistTreeNode>>,
64        id: PlaylistTreeNodeId,
65    ) -> impl Iterator<Item = crate::Result<PlaylistNode>> + '_ {
66        playlists
67            .get(&id)
68            .into_iter()
69            .flat_map(|nodes| nodes.iter())
70            .map(|node| -> crate::Result<PlaylistNode> {
71                let child_node = if node.is_folder() {
72                    let folder = PlaylistFolder {
73                        id: node.id,
74                        name: node.name.clone().into_string()?,
75                        children: get_child_nodes(playlists, node.id)
76                            .collect::<crate::Result<Vec<PlaylistNode>>>()?,
77                    };
78                    PlaylistNode::Folder(folder)
79                } else {
80                    let playlist = Playlist {
81                        id: node.id,
82                        name: node.name.clone().into_string()?,
83                    };
84                    PlaylistNode::Playlist(playlist)
85                };
86                Ok(child_node)
87            })
88    }
89
90    get_child_nodes(&playlists, PlaylistTreeNodeId(0)).collect::<crate::Result<Vec<PlaylistNode>>>()
91}
